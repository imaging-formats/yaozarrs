yaozarrs.typing #

Types used in the signatures of yaozarrs' public API.

Use these to annotate values that you pass to yaozarrs, or get back from it. They exist only for type checkers, so import them under if TYPE_CHECKING: (with from __future__ import annotations, or quoted annotations, on Python < 3.14):

from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from yaozarrs.typing import CompressionName

compression: CompressionName = "zstd"

Since that import never runs, it works with any version of yaozarrs at runtime (including those that predate this module), so it needs no minimum version.

Removing or renaming a name here is a breaking change. A Literal may gain values (e.g. a new compression codec) in a minor release.

CreateArrayFunc #

Protocol for custom array creation functions.

This is the type signature for functions that can be passed as the writer parameter to many functions in this module. Use this if you'd like to completely customize how arrays are created, e.g., using a different backend, or custom storage options. See _create_array_zarr and _create_array_tensorstore for (our internal) reference implementations.

def __init__(
    self,
    store: Mapping[str, bytes],
    path: str = "",
    meta: ZarrMetadata | None = None,
) -> None:
    """Initialize a zarr node.

    Parameters
    ----------
    store : Mapping[str, bytes]
        The mapping for the zarr store (e.g. an FSMap, or a dict).  Anything other
        than a _CachedMapper will be wrapped in one, for performance.
    path : str
        The path to this node within the zarr store (relative to the store root).
    meta : dict[str, Any] | ZarrMetadata | None
        Optional pre-loaded metadata dictionary. If not provided, it will be
        loaded from the store.
    """
    # Ensure we have a cached mapper for performance
    if not isinstance(store, _CachedMapper):
        store = _CachedMapper(store)

    self._store = store
    self._path = str(path).rstrip("/")
    if meta is None:
        self._metadata = _load_zarr_metadata(self._store, self._path)
    elif isinstance(meta, ZarrMetadata):
        if meta.node_type != self.node_type():  # pragma: no cover
            raise ValueError(
                f"Metadata node_type '{meta.node_type}' does not match "
                f"expected '{self.node_type()}'"
            )
        self._metadata = meta
    else:  # pragma: no cover
        raise ValueError("meta must be a ZarrMetadata instance or None")

def to_tensorstore(self) -> tensorstore.TensorStore:
    """Convert to a tensorstore TensorStore object.

    !!!important
        Requires `tensorstore` to be installed.
    """
    try:
        import tensorstore as ts
    except ImportError as e:
        raise ImportError(
            "tensorstore package is required for to_tensorstore()"
        ) from e

    fsmap, *_ = self._full_path()
    spec = {
        "driver": "zarr3" if self._metadata.zarr_format == 3 else "zarr",
        "kvstore": _fsmap_to_tensorstore_kvstore(fsmap, self._path),
    }
    future = ts.open(spec)
    return future.result()

def ome_metadata(
    self, *, version: str | None = None, strict: bool | None = None
) -> v06.OMEMetadata | v05.OMEMetadata | v04.OMEZarrGroupJSON | None:
    """Return the OME metadata if present in attributes, else None."""
    attrs = self.attributes
    version = version or self._guess_ome_version()
    if version in (*V06_VERSIONS, "0.5") and "ome" not in attrs:
        return None
    if version in V06_VERSIONS:
        return type_adapter(v06.OMEMetadata).validate_python(
            attrs["ome"], strict=strict
        )
    elif version == "0.5":
        return type_adapter(v05.OMEMetadata).validate_python(
            attrs["ome"], strict=strict
        )
    elif version == "0.4":
        return type_adapter(v04.OMEZarrGroupJSON).validate_python(
            attrs, strict=strict
        )
    elif version:
        raise ValueError(f"Unsupported OME-Zarr version: {version}")
    elif "bioformats2raw.layout" in attrs:
        # this is the most annoying case... something like
        # {'bioformats2raw.layout': 3} in ome zarr v0.4 ... there's no way
        # to determine version for sure without traversing the tree.
        # so we just try to fallback to v4 if no version has been specified/parsed.
        with suppress(ValidationError):
            return type_adapter(v04.Bf2Raw).validate_python(attrs, strict=strict)
    return None
